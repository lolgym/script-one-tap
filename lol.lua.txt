-- UNLOAD
local function Unload()
    scriptLoaded = false
    if fovCircle then fovCircle:Destroy() fovCircle = nil end
    if gui then gui:Destroy() gui = nil end
    if mobileButton then mobileButton:Destroy() mobileButton = nil end
    if updateConn then updateConn:Disconnect() end
    if flyConn then flyConn:Disconnect() end
    
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChild("Humanoid")
        if hum then
            hum.PlatformStand = false
            hum.Jump = true
        end
    end
    
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character then
            for _, o in ipairs(p.Character:GetChildren()) do
                if o:IsA("BoxHandleAdornment") or o:IsA("BillboardGui") then
                    o:Destroy()
                end
            end
        end
    end
    for _, c in ipairs(workspace:GetChildren()) do
        if c:IsA("Model") and c:FindFirstChild("Humanoid") then
            for _, o in ipairs(c:GetChildren()) do
                if o:IsA("BoxHandleAdornment") or o:IsA("BillboardGui") then
                    o:Destroy()
                end
            end
        end
    end
    print("Unloaded")
end

-- CENTER
local function GetCenter()

    return Vector2.new(size.X/2 + CENTER_OFFSET.X, size.Y/2 + CENTER_OFFSET.Y)
end

-- FOV CIRCLE
local function CreateFOVCircle()
    if fovCircle then fovCircle:Destroy() fovCircle = nil end
    local center = GetCenter()
    local r = FOV
    local c = Instance.new("ScreenGui")
    c.Parent = LocalPlayer.PlayerGui
    c.ResetOnSpawn = false
    c.Name = "FOVCircle"
    c.Enabled = false
    
    local f = Instance.new("Frame")
    f.Size = UDim2.new(0, r*2, 0, r*2)
    f.Position = UDim2.new(0, center.X - r, 0, center.Y - r)
    f.BackgroundTransparency = 1
    f.Parent = c
    f.Name = "CircleFrame"
    
    for i = 0, 63 do
        local a = (i/64) * 2 * math.pi
        local na = ((i+1)/64) * 2 * math.pi
        local x1, y1 = r*math.cos(a), r*math.sin(a)
        local x2, y2 = r*math.cos(na), r*math.sin(na)
        local mx, my = (x1+x2)/2, (y1+y2)/2
        local len = math.sqrt((x2-x1)^2 + (y2-y1)^2)
        local rot = math.atan2(y2-y1, x2-x1) * 180 / math.pi
        
        local seg = Instance.new("Frame")
        seg.Size = UDim2.new(0, len, 0, 2)
        seg.Position = UDim2.new(0.5, mx - len/2, 0.5, my - 1)
        seg.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
        seg.BackgroundTransparency = 0.4
        seg.BorderSizePixel = 0
        seg.Rotation = rot
        seg.Parent = f
    end
    
    local dot = Instance.new("Frame")
    dot.Size = UDim2.new(0, 3, 0, 3)
    dot.Position = UDim2.new(0.5, -1.5, 0.5, -1.5)
    dot.BackgroundColor3 = Color3.fromRGB(0, 255, 100)
    dot.BackgroundTransparency = 0.2
    dot.BorderSizePixel = 0
    dot.Parent = f
    
    fovCircle = c
    return c
end

local function UpdateFOVCircle()
    if not fovCircle then return end
    local f = fovCircle:FindFirstChild("CircleFrame")
    if not f then return end
    local center = GetCenter()
    local r = FOV
    f.Size = UDim2.new(0, r*2, 0, r*2)
    f.Position = UDim2.new(0, center.X - r, 0, center.Y - r)
    local children = f:GetChildren()
    for i = 0, 63 do
        local a = (i/64) * 2 * math.pi
        local na = ((i+1)/64) * 2 * math.pi
        local x1, y1 = r*math.cos(a), r*math.sin(a)
        local x2, y2 = r*math.cos(na), r*math.sin(na)
        local mx, my = (x1+x2)/2, (y1+y2)/2
        local len = math.sqrt((x2-x1)^2 + (y2-y1)^2)
        local rot = math.atan2(y2-y1, x2-x1) * 180 / math.pi
        local seg = children[i+1]
        if seg and seg:IsA("Frame") then
            seg.Size = UDim2.new(0, len, 0, 2)
            seg.Position = UDim2.new(0.5, mx - len/2, 0.5, my - 1)
            seg.Rotation = rot
        end
    end
end

-- NPC
local function IsNPC(c)
    if not c then return false end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character == c then return false end
    end
    return true
end

local function GetName(c)
    if IsNPC(c) then return "NPC" end
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character == c then return p.Name end
    end
    return "?"
end

local function IsVisible(c)
    if not CheckWalls then return true end
    local r = c:FindFirstChild("HumanoidRootPart")
    if not r then return false end
    local o = Camera.CFrame.Position
    local t = r.Position + Vector3.new(0, 1.5, 0)
    local ray = Ray.new(o, (t - o).Unit * (t - o).Magnitude)
    local h = workspace:FindPartOnRay(ray, LocalPlayer.Character)
    if h then
        local p = h.Parent
        if p == c or p:IsDescendantOf(c) then return true end
        return false
    end
    return true
end

local function GetTargets()
    local t = {}
    for _, p in ipairs(Players:GetPlayers()) do
        local c = p.Character
        if c and c ~= LocalPlayer.Character and c:FindFirstChild("HumanoidRootPart") and c:FindFirstChild("Humanoid") and c.Humanoid.Health > 0 then
            table.insert(t, c)
        end
    end
    for _, c in ipairs(workspace:GetChildren()) do
        if c:IsA("Model") and c:FindFirstChild("Humanoid") and c:FindFirstChild("HumanoidRootPart") and c ~= LocalPlayer.Character then
            local isP = false
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Character == c then isP = true break end
            end
            if not isP and c.Humanoid.Health > 0 then
                table.insert(t, c)
            end
        end
    end
    return t
end

local function GetClosest()
    local targets = GetTargets()
    local closest = nil
    local dist = math.huge
    local center = Vector2.new(Mouse.X, Mouse.Y)
    for _, t in ipairs(targets) do
        local r = t:FindFirstChild("HumanoidRootPart")
        if r then
            local pos, on = Camera:WorldToScreenPoint(r.Position)
            if on then
                local d = (Vector2.new(pos.X, pos.Y) - center).Magnitude
                if d < dist and d < FOV and IsVisible(t) then
                    dist = d
                    closest = t
                end
            end
        end
    end
    return closest
end

-- FLY
local function StartFlyLoop()
    if flyConn then flyConn:Disconnect() end
    flyConn = RunService.Heartbeat:Connect(function()
        if not scriptLoaded or not Fly then return end
        local char = LocalPlayer.Character
        if not char then return end
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        
        local move = Vector3.new(0, 0, 0)
        local spd = 50 * Speed
        
        if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + Camera.CFrame.LookVector * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - Camera.CFrame.LookVector * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - Camera.CFrame.RightVector * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + Camera.CFrame.RightVector * spd end
        if UserInputService:IsKeyDown(Enum.KeyCode.Space) then move = move + Vector3.new(0, spd, 0) end
        if UserInputService:IsKeyDown(Enum.KeyCode.LeftShift) then move = move - Vector3.new(0, spd, 0) end
        
        if move.Magnitude > 0 then
            hrp.CFrame = hrp.CFrame + move * 0.06
        end
        
        local hum = char:FindFirstChild("Humanoid")
        if hum then hum.PlatformStand = true end
    end)
end

local function StartUpdateLoop()
    if updateConn then updateConn:Disconnect() end
    updateConn = RunService.Heartbeat:Connect(function()
        if not scriptLoaded then return end
        local char = LocalPlayer.Character
        if not char then return end
        
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChild("Humanoid")
        if not hrp or not hum then return end
        
        if NoClip then
            for _, part in ipairs(char:GetDescendants()) do
                if part:IsA("BasePart") then
                    part.CanCollide = false
                end
            end
        end
        
        hum.WalkSpeed = 16 * Speed
        hum.JumpPower = 50 * Speed
        
        if Fly then
            hum.PlatformStand = true
            hrp.Anchored = false
        else
            hum.PlatformStand = false
            hrp.Anchored = false
        end
    end)
end

-- RESPAWN
local function ForceRespawn()
    if respawning then return end
    respawning = true
    
    local char = LocalPlayer.Character
    if char then
        local hrp = char:FindFirstChild("HumanoidRootPart")
        local pos = hrp and hrp.Position or Vector3.new(0, 10, 0)
        
        local hum = char:FindFirstChild("Humanoid")
        if hum then
            hum.Health = 0
        end
        
        task.wait(0.5)
        
        local newChar = LocalPlayer.Character
        if newChar then
            local newHrp = newChar:FindFirstChild("HumanoidRootPart")
            if newHrp then
                newHrp.CFrame = CFrame.new(pos)
            end
            local newHum = newChar:FindFirstChild("Humanoid")
            if newHum then
                newHum.WalkSpeed = 16 * Speed
                newHum.JumpPower = 50 * Speed
                newHum.Jump = true
                newHum.PlatformStand = false
            end
        end
    end
    
    respawning = false
end

local function ToggleFly()
    Fly = not Fly
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChild("Humanoid")
    
    if Fly then
        StartFlyLoop()
        if hum then 
            hum.PlatformStand = true
            hum.Jump = true
        end
        print("FLY ON")
    else
        if flyConn then flyConn:Disconnect() flyConn = nil end
        if hum then 
            hum.PlatformStand = false
            hum.WalkSpeed = 16 * Speed
            hum.JumpPower = 50 * Speed
            hum.Jump = true
        end
        print("FLY OFF - Respawn...")
        ForceRespawn()
    end
end

    local function UpdateContent(tabName)
        for _, child in ipairs(content:GetChildren()) do
            child:Destroy()
        end
        
        local y = 0
        
        if tabName == "PLAYER" then
            CreateSection(content, "PLAYER DETAILS", y)
            y = y + 0.07
            
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local pos = hrp and hrp.Position or Vector3.new(0,0,0)
            
            MakeInfo(content, "Player Name:", LocalPlayer.Name, y)
            y = y + 0.055
            MakeInfo(content, "Position:", string.format("%.0f, %.0f, %.0f", pos.X, pos.Y, pos.Z), y)
            y = y + 0.055
            MakeInfo(content, "Health:", char and char:FindFirstChild("Humanoid") and math.floor(char.Humanoid.Health) or "N/A", y)
            y = y + 0.055
            MakeInfo(content, "Speed:", Speed .. "x", y)
            y = y + 0.055
            MakeInfo(content, "Players Online:", #Players:GetPlayers(), y)
            y = y + 0.055
            
            CreateSection(content, "DUTY STATUS", y)
            y = y + 0.07
            
            MakeInfo(content, "Status:", Fly and "FLYING" or "GROUND", y)
            y = y + 0.055
            MakeInfo(content, "NoClip:", NoClip and "ENABLED" or "DISABLED", y)
            y = y + 0.055
            MakeInfo(content, "Aimbot:", AIMon and "ACTIVE" or "INACTIVE", y)
            y = y + 0.055
            MakeInfo(content, "ESP:", ESPon and "ON" or "OFF", y)
            
        elseif tabName == "COMBAT" then
            CreateSection(content, "COMBAT SETTINGS", y)
            y = y + 0.07
            
            MakeToggle(content, "AIMBOT", y, function() return AIMon end, function(v) AIMon = v ToggleFOVCircle(AIMon) end)
            y = y + 0.065
            MakeToggle(content, "WALL CHECK", y, function() return CheckWalls end, function(v) CheckWalls = v end)
            y = y + 0.065
            MakeToggle(content, "FOV CIRCLE", y, function() return ShowFOVCircle end, function(v) ShowFOVCircle = v ToggleFOVCircle(AIMon and ShowFOVCircle) end)
            y = y + 0.065
            MakeSlider(content, "FOV", y, 50, 500, function() return FOV end, function(v) FOV = v UpdateFOVCircle() end)
            y = y + 0.095
            MakeSlider(content, "SMOOTHNESS", y, 1, 100, function() return math.floor(Smoothness * 100) end, function(v) Smoothness = v/100 end)
            
        elseif tabName == "VISUALS" then
            CreateSection(content, "VISUAL SETTINGS", y)
            y = y + 0.07
            
            MakeToggle(content, "ESP", y, function() return ESPon end, function(v)
                ESPon = v
                if not v then
                    for _, p in ipairs(Players:GetPlayers()) do if p.Character then RemoveESP(p.Character) end end
                    for _, c in ipairs(workspace:GetChildren()) do if c:IsA("Model") and c:FindFirstChild("Humanoid") then RemoveESP(c) end end
                else
                    for _, p in ipairs(Players:GetPlayers()) do if p.Character then CreateESP(p.Character) end end
                    for _, c in ipairs(workspace:GetChildren()) do if c:IsA("Model") and c:FindFirstChild("Humanoid") and c ~= LocalPlayer.Character then CreateESP(c) end end
                end
            end)
            y = y + 0.065
            MakeSlider(content, "BOX SIZE", y, 1, 10, function() return math.floor(BoxSize) end, function(v)
                BoxSize = v
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character then
                        local b = p.Character:FindFirstChild("BoxHandleAdornment")
                        if b then b.Size = Vector3.new(BoxSize, BoxSize + 1.5, 1) end
                    end
                end
                for _, c in ipairs(workspace:GetChildren()) do
                    if c:IsA("Model") and c:FindFirstChild("Humanoid") then
                        local b = c:FindFirstChild("BoxHandleAdornment")
                        if b then b.Size = Vector3.new(BoxSize, BoxSize + 1.5, 1) end
                    end
                end
            end)
            y = y + 0.095
            MakeSlider(content, "TEXT SIZE", y, 8, 24, function() return ESPsize end, function(v)
                ESPsize = v
                for _, p in ipairs(Players:GetPlayers()) do
                    if p.Character then
                        local bg = p.Character:FindFirstChild("BillboardGui")
                        if bg then
                            local lbl = bg:FindFirstChild("TextLabel")
                            if lbl then lbl.TextSize = v end
                        end
                    end
                end
            end)
            
        elseif tabName == "FUTURES" then
            CreateSection(content, "FUTURES", y)
            y = y + 0.07
            
            MakeToggle(content, "FLY [F]", y, function() return Fly end, function(v) ToggleFly() end)
            y = y + 0.065
            MakeToggle(content, "NOCLIP", y, function() return NoClip end, function(v) ToggleNoClip() end)
            y = y + 0.065
            MakeSlider(content, "SPEED", y, 1, 5, function() return Speed end, function(v) SetSpeed(v) end)
        end
        
        -- Добавляем отступ снизу
        local spacer = Instance.new("Frame")
        spacer.Size = UDim2.new(1, 0, 0, 10)
        spacer.Position = UDim2.new(0, 0, y + 0.02, 0)
        spacer.BackgroundTransparency = 1
        spacer.Parent = content
    end
    
    -- Вкладки
    local tabs = {"PLAYER", "COMBAT", "VISUALS", "FUTURES"}
    local tabBtns = {}
    
    for i, name in ipairs(tabs) do
        local btn = Instance.new("TextButton")
        btn.Size = UDim2.new(0.25, 0, 1, 0)
        btn.Position = UDim2.new((i-1) * 0.25, 0, 0, 0)
        btn.BackgroundTransparency = 1
        btn.Text = name
        btn.TextColor3 = (i == 1) and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(150, 150, 170)
        btn.TextSize = 10
        btn.Font = Enum.Font.GothamBold
        btn.Parent = tabBar
        btn.Name = name
        tabBtns[name] = btn
        
        btn.MouseButton1Click:Connect(function()
            for _, b in pairs(tabBtns) do
                b.TextColor3 = Color3.fromRGB(150, 150, 170)
            end
            btn.TextColor3 = Color3.fromRGB(255, 255, 255)
            UpdateContent(name)
        end)
    end
    
    UpdateContent("PLAYER")
    return gui
end

local function ScanAll()
    for _, p in ipairs(Players:GetPlayers()) do
        if p.Character and p.Character ~= LocalPlayer.Character then
            CreateESP(p.Character)
        end
    end
    for _, c in ipairs(workspace:GetChildren()) do
        if c:IsA("Model") and c:FindFirstChild("Humanoid") and c ~= LocalPlayer.Character then
            local isP = false
            for _, p in ipairs(Players:GetPlayers()) do
                if p.Character == c then isP = true break end
            end
            if not isP then CreateESP(c) end
        end
    end
end

UserInputService.InputBegan:Connect(function(input, gp)
    if gp then return end
    if not scriptLoaded then return end
    
    if input.KeyCode == Enum.KeyCode.X then
        menuOpen = not menuOpen
        if gui then
            local m = gui:FindFirstChild("Main")
            if m then m.Visible = menuOpen end
        end
    end
    
    if input.KeyCode == Enum.KeyCode.Z then
        AIMon = not AIMon
        ToggleFOVCircle(AIMon and ShowFOVCircle)
    end
    
    if input.KeyCode == Enum.KeyCode.F then
        ToggleFly()
        if gui and gui:FindFirstChild("Main") and gui.Main:FindFirstChild("Content") then
            local content = gui.Main.Content
            for _, child in ipairs(content:GetChildren()) do
                if child:IsA("TextButton") then
                    local lbl = child:FindFirstChild("TextLabel")
                    if lbl and lbl.Text == "FLY [F]" then
                        local st = child:FindFirstChildOfClass("TextLabel")
                        if st and st.TextXAlignment == Enum.TextXAlignment.Right then
                            st.Text = Fly and "ON" or "OFF"
                            st.TextColor3 = Fly and Color3.fromRGB(0, 255, 100) or Color3.fromRGB(255, 80, 80)
                        end
                    end
                end
            end
        end
    end
end)

RunService.Heartbeat:Connect(function()
    if not scriptLoaded or not AIMon then return end
    local target = GetClosest()
    if target then
        local root = target:FindFirstChild("HumanoidRootPart")
        if root then
            local pos = root.Position + Vector3.new(0, 1.5, 0)
            local current = Camera.CFrame.Position
            local look = (pos - current).Unit
            local newLook = Camera.CFrame.LookVector:Lerp(look, Smoothness)
            Camera.CFrame = CFrame.lookAt(current, current + newLook)
        end
    end
end)
